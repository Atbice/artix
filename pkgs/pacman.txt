# Packages from official Artix + Arch repos.
# One per line, '#' comments and blank lines ignored. Installed by bootstrap.sh
# via `pacman -S --needed`. Multilib must be enabled in /etc/pacman.conf —
# bootstrap.sh does that before this list is installed.

# --- base system ------------------------------------------------------------
# (Assumes you installed Artix base-dinit from the ISO; these are top-ups.)
base-devel              # makepkg for AUR
git
curl
chrony
chrony-dinit            # Artix dinit service file for chrony
dinit-rc                # dinit boot scripts metapackage (no-op if already there)

# --- networking + audio + session services ----------------------------------
networkmanager
networkmanager-dinit
dbus-dinit
# Session + seat management. Chosen over elogind because elogind is a
# systemd code fork (independent maintainers, but the source lineage is
# systemd's logind). turnstile + seatd are from-scratch implementations
# with no systemd DNA. See docs/adr/0002. Trade-off: turnstile is pre-1.0
# (0.1.x) and quieter upstream than elogind. Swap-back is 5 lines.
seatd                   # low-level seat manager (/dev/dri, /dev/input access)
seatd-dinit
turnstile               # session manager + logind D-Bus compatibility shim
turnstile-dinit
pipewire
pipewire-pulse          # PulseAudio replacement
pipewire-alsa
wireplumber
xdg-utils
polkit
polkit-gnome            # generic polkit agent; works under labwc

# --- display server / portals -----------------------------------------------
# labwc is Wayland-native (wlroots-based); no Xorg server needed for the
# session, but xorg-xwayland is required for any X11 client (most games,
# older apps).
xorg-xwayland
xdg-desktop-portal
xdg-desktop-portal-wlr    # screencast + screen-share for wlroots compositors
xdg-desktop-portal-gtk    # GTK file pickers fallback

# --- labwc + greetd ---------------------------------------------------------
labwc                   # Wayland stackable compositor (wlroots, Openbox-style config)
greetd
greetd-dinit            # Artix dinit service file for greetd
greetd-tuigreet         # TUI greeter (lightweight, terminal-based)

# --- Noctalia v5 build deps (source-built by bootstrap.sh) ------------------
# v5 is a native Wayland/GLES shell, NOT QML/Quickshell. Pacman doesn't have
# it yet (only v4 is on AUR), so bootstrap.sh clones the v5 branch and runs
# `just configure release && just build release && sudo just install release`.
# These are the upstream Arch deps from the v5 README:
meson
just                    # command runner used by the v5 Justfile
wayland-protocols
libglvnd                # libGL/libEGL umbrella
freetype2
fontconfig
cairo
pango
harfbuzz
libxkbcommon
glib2
sdbus-cpp
libpipewire
pam
libwebp
librsvg

# --- terminal + small utilities ---------------------------------------------
foot                    # tiny, fast Wayland-native terminal
fuzzel                  # app launcher fallback (Noctalia provides its own,
                        # but fuzzel is handy as a backup / for scripts)
mako                    # notification daemon fallback (Noctalia provides
                        # notifications; mako kept as a tested fallback)
swaylock                # screen lock for Wayland
swayidle                # idle daemon for auto-lock / dpms
wl-clipboard            # wl-copy / wl-paste
grim                    # screenshot
slurp                   # region select (pairs with grim)
brightnessctl           # backlight control (laptops; harmless on desktop)
playerctl               # media-key control for media players
pavucontrol             # PipeWire/Pulse mixer GUI

# --- file management --------------------------------------------------------
nautilus                # GTK file manager — works under labwc, no KDE deps

# --- nvidia (RTX 3090, Ampere → current branch, DKMS) -----------------------
linux-headers           # required for DKMS (mainline kernel)
linux-lts-headers       # so nvidia-dkms also builds for linux-lts (fallback kernel)
nvidia-dkms
nvidia-utils
lib32-nvidia-utils      # MULTILIB — mandatory for Steam/Proton games
nvidia-settings
libva-nvidia-driver     # VA-API → NVDEC bridge (video acceleration)

# --- gaming -----------------------------------------------------------------
steam                   # from [multilib]
lutris                  # from [extra]
gamemode                # gamemoderun wrapper, CPU governor tweaks
lib32-gamemode
mangohud
lib32-mangohud
wine
winetricks

# --- fonts ------------------------------------------------------------------
noto-fonts
noto-fonts-emoji
noto-fonts-cjk          # CJK coverage
ttf-dejavu
ttf-liberation
ttf-jetbrains-mono      # nice mono for foot
